function whyTxt(S,b){const v=S.rt.v,st=S.rt.st[b.id]||{},P=b.p||{},on=x=>x>.5,val=n=>S.nets[n].dig?(on(v[n])?'1':'0'):fmt(v[n]),ins=b.pins.filter(p=>p.role==='in'),

  case 'AND':{const z=ins.filter(p=>!on(v[p.n]));return o==='1'?'AND = 1: ALL inputs are 1.':'AND = 0: '+(z.length?lst(z)+' is 0 (one 0 is enough).':'an input is 0.')}
  case 'OR':{const z=ins.filter(p=>on(v[p.n]));return o==='1'?'OR = 1: '+lst(z)+' is 1 (one 1 is enough).':'OR = 0: ALL inputs are 0.'}
  case 'NOT':return 'NOT = '+o+': the input is '+(ins[0]?val(ins[0].n):'?')+', the output is the opposite.';
